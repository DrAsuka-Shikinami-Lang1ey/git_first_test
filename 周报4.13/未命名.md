# 应用  
  
一款串口控制的智能风扇  
  
- 温度采集  
- 指示灯刷新  
- 电机控制  
- 按钮事件  
  
  
  
# 事件  
  
- adc_ntc_get  
- led_set  
- motor_set  
- key_set  
  
# 制定指标  
  
运行速度不是越快越好，而是合适最好  
  
- 温度：1s  
- LED：20ms  
- 马达：10ms  
- 按钮：10ms  
  
有一个系统定时器 systick，值是每 1ms 加一  
  
## 时间片轮转算法  
  
```c  
  
void fun_adc();  
void fun_led();  
void fun_motor();  
void fun_key();  
  
main  
{  
        ti_adc = systick;  
        ti_led = systick;  
        ti_motor = systick;  
        ti_key = systick;  
          
        while(1)  
        {  
                if (systick - ti_adc  >= 1000) {  
                        fun_adc();  
                        ti_adc = systick;  
                }  
                if (systick - ti_led  >= 20) {  
                        fun_led();  
                        ti_led = systick;  
                }  
                if (systick - ti_motor  >= 10 ) {  
                        fun_motor();  
                        ti_motor = systick;  
                }  
                if (systick - ti_key  >= 10 ) {  
                        fun_key();  
                        ti_key = systick;  
                }          
        }  
          
}  
  
  
```  
  
## 时间片调度表算法  
  
  
```c  
void fun_adc();  
void fun_led();  
void fun_motor();  
void fun_key();  
  
typedef struct  
{  
        void (*task)(); /**< 任务指针 **/  
        uint32_t period;  /**< 执行周期 **/  
        uint32_t start_ti /**< 开始时间戳 **/  
}task_t;  
  
task_t task_table[] = [  
        {fun_adc, 1000, 0},  
        {fun_led, 20, 0},  
        {fun_motor, 10, 0},  
        {fun_key, 10, 0},  
        {NULL}  
]  
  
  
int main()  
{  
        while(1)  
        {  
                for(uint8_t idx = 0; task_table[idx].task != NULL; idx++)  
                {  
                        if (systick - task_table[idx].start_ti >= task_table[idx].period)  
                        {  
                                task_table[idx].task();  
                                task_table[idx].start_ti = systick;  
                        }  
                }  
        }  
  
}  
  
  
```
